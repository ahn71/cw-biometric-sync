    def __decode_time(self, t):        
        raw = t.hex()

        t_int = unpack("<I", t)[0]

        # print("\nRAW:", raw)
        # print("INT:", t_int)

        second = t_int % 60
        t_int //= 60

        minute = t_int % 60
        t_int //= 60

        hour = t_int % 24
        t_int //= 24

        day = t_int % 31 + 1
        t_int //= 31

        month = t_int % 12 + 1
        t_int //= 12

        year = t_int + 2000

        # print(
        #     f"DECODE => {year}-{month}-{day} "
        #     f"{hour}:{minute}:{second}"
        # )

        return datetime(year, month, day, hour, minute, second)
